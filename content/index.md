<script>
function updateTime() {
  const now = new Date();
  const options = {
    year: 'numeric', month: 'long', day: 'numeric',
    hour: '2-digit', minute: '2-digit', second: '2-digit',
    hour12: false
  };
  document.getElementById('current-time').textContent =
    '📅 ' + now.toLocaleString('zh-CN', options);
}
updateTime();
setInterval(updateTime, 1000);
</script>
